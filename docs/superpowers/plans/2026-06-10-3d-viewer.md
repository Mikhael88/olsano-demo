# Olsano 3D Viewer — Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Build a single `index.html` file that renders `watering-can.glb` with 4 PBR material presets and a real-time albedo color picker, styled for the Olsano brand.

**Architecture:** One self-contained `index.html`. ES module importmap loads Three.js r170 from `esm.sh` CDN. `WebGPURenderer` with automatic WebGL2 fallback. All HTML, CSS, and JS inlined. Must be served over HTTP (not `file://`) — use VS Code Live Server, `npx serve .`, or `python -m http.server 8080`.

**Tech Stack:** Three.js r170 (esm.sh CDN) · WebGPURenderer · GLTFLoader · DRACOLoader · OrbitControls · RoomEnvironment · PMREMGenerator · Google Fonts (Cormorant Garamond + Outfit)

```js
// ─── Material presets ─────────────────────────────────────────────────────────
const MATERIAL_PRESETS = {
  matte: {
    roughness: 0.85,
    metalness: 0.0,
  },
  glass: {
    roughness: 0.05,
    metalness: 0.0,
    transmission: 0.92,
    thickness: 0.5,
    ior: 1.5,
    transparent: true,
    side: THREE.DoubleSide,
  },
  polished: {
    roughness: 0.08,
    metalness: 0.0,
    clearcoat: 1.0,
    clearcoatRoughness: 0.05,
  },
  terracotta: {
    roughness: 0.90,
    metalness: 0.0,
  },
};

// Colors locked per preset — ignores activeColor when set
const LOCKED_COLORS = {
  terracotta: '#b5611a',
};

function applyMaterial(matKey, hexColor) {
  if (!gltfScene) return;
  const preset     = MATERIAL_PRESETS[matKey];
  const colorValue = LOCKED_COLORS[matKey] ?? hexColor;

  gltfScene.traverse((node) => {
    if (!node.isMesh) return;
    if (node.material) {
      Array.isArray(node.material)
        ? node.material.forEach(m => m.dispose())
        : node.material.dispose();
    }
    node.material = new THREE.MeshPhysicalMaterial({
      color: new THREE.Color(colorValue),
      envMapIntensity: 1.0,
      ...preset,
    });
  });
}
```

```js
// ─── Camera auto-fit ─────────────────────────────────────────────────────────
function fitCamera(object) {
  const box    = new THREE.Box3().setFromObject(object);
  const size   = box.getSize(new THREE.Vector3());
  const center = box.getCenter(new THREE.Vector3());

  const maxDim = Math.max(size.x, size.y, size.z);
  const fov    = camera.fov * (Math.PI / 180);
  const dist   = Math.abs(maxDim / (2 * Math.tan(fov / 2))) * 1.6;

  camera.position.set(center.x, center.y + size.y * 0.1, center.z + dist);
  camera.near = dist / 100;
  camera.far  = dist * 10;
  camera.updateProjectionMatrix();

  controls.target.copy(center);
  controls.update();
}

// ─── GLB Loader ──────────────────────────────────────────────────────────────
const dracoLoader = new DRACOLoader();
dracoLoader.setDecoderPath('https://www.gstatic.com/draco/versioned/decoders/1.5.6/');

const gltfLoader = new GLTFLoader();
gltfLoader.setDRACOLoader(dracoLoader);

gltfLoader.load(
  './.glb-file/watering-can.glb',
  (gltf) => {
    gltfScene = gltf.scene;
    scene.add(gltfScene);
    fitCamera(gltfScene);
    applyMaterial(activeMat, activeColor);

    // Hide loader, show controls + hint
    loaderEl.hidden = true;
    controlsEl.classList.add('visible');
    hintEl.classList.add('visible');
  },
  undefined,
  (_err) => {
    loaderEl.hidden = true;
    errorEl.hidden = false;
  }
);
```

```js
// ─── Ground shadow ────────────────────────────────────────────────────────────
function createGroundShadow(object) {
  const box    = new THREE.Box3().setFromObject(object);
  const center = box.getCenter(new THREE.Vector3());
  const size   = box.getSize(new THREE.Vector3());
  const radius = Math.max(size.x, size.z) * 0.7;

  // Build radial gradient texture on a canvas
  const res = 256;
  const shadowCanvas = document.createElement('canvas');
  shadowCanvas.width = shadowCanvas.height = res;
  const ctx = shadowCanvas.getContext('2d');
  const gradient = ctx.createRadialGradient(res/2, res/2, 0, res/2, res/2, res/2);
  gradient.addColorStop(0,   'rgba(0,0,0,0.22)');
  gradient.addColorStop(0.6, 'rgba(0,0,0,0.08)');
  gradient.addColorStop(1,   'rgba(0,0,0,0)');
  ctx.fillStyle = gradient;
  ctx.fillRect(0, 0, res, res);

  const texture = new THREE.CanvasTexture(shadowCanvas);
  const mesh = new THREE.Mesh(
    new THREE.CircleGeometry(radius, 48),
    new THREE.MeshBasicMaterial({
      map: texture,
      transparent: true,
      depthWrite: false,
      side: THREE.DoubleSide,
    })
  );

  mesh.rotation.x = -Math.PI / 2;
  mesh.position.set(center.x, box.min.y + 0.001, center.z);
  return mesh;
}
```
